// Tinkers: auto-render products from products.json (GitHub Pages friendly) const PRODUCTS_URL = 'products.json'; const CART_KEY = 'tinkers_cart_v1'; function moneyZAR(value){ const n = Number(value || 0); return 'R' + n.toLocaleString('en-ZA'); } function getImage(p){ return (p && (p.image || p.filename)) ? ('images/' + (p.image || p.filename)) : ''; } function readCart(){ try { return JSON.parse(localStorage.getItem(CART_KEY) || '[]'); } catch { return []; } } function writeCart(cart){ localStorage.setItem(CART_KEY, JSON.stringify(cart)); } function addToCart(id){ const cart = readCart(); const found = cart.find(x => x.id === id); if(found) found.qty += 1; else cart.push({id, qty: 1}); writeCart(cart); updateCartCount(); // gentle feedback alert('Added to cart'); // if cart panel exists, refresh it renderCartPanel(window.__products || []); } function updateQty(id, delta){ let cart = readCart(); cart = cart.map(x => x.id === id ? ({...x, qty: x.qty + delta}) : x).filter(x => x.qty > 0); writeCart(cart); updateCartCount(); renderCartPanel(window.__products || []); renderCheckout(window.__products || []); } function updateCartCount(){ const el = document.getElementById('cartCount'); if(!el) return; const count = readCart().reduce((s,x)=>s+x.qty,0); el.textContent = String(count); } function openChat(){ alert('AI Curator: Ask me about product stories, gifting, and recommendations'); } async function loadProducts(){ const res = await fetch(PRODUCTS_URL, {cache: 'no-store'}); if(!res.ok) throw new Error('Could not load products.json'); const data = await res.json(); // support both formats: array or {products:[...]} const products = Array.isArray(data) ? data : (data.products || []); window.__products = products; return products; } function wireNavFilters(products){ document.querySelectorAll('nav a[data-filter]').forEach(a => { a.addEventListener('click', (e) => { const cat = a.getAttribute('data-filter'); renderHome(products, cat); }); }); } function renderHome(products, category='All'){ const grid = document.getElementById('products'); if(!grid) return; // Featured Original Art const originals = products.filter(p => (p.type || '').toLowerCase() === 'original'); const featured = document.getElementById('originalArt'); if(featured){ if(originals.length){ const p = originals[0]; featured.style.display = 'block'; featured.innerHTML = `
Original · Signed

${p.name}

By ${p.artist || 'Artist'}

Medium: ${p.medium || 'Canvas'} Edition: ${p.edition || 'One-of-one'}
${moneyZAR(p.price)}
View Artwork Details
`; } else { featured.style.display = 'none'; featured.innerHTML = ''; } } // Retail catalogue const list = products.filter(p => (p.type || '').toLowerCase() !== 'original'); const filtered = category === 'All' ? list : list.filter(p => p.category === category); grid.innerHTML = filtered.map(p => { return `

${p.name}

${p.category}

${moneyZAR(p.price)}

View
`; }).join(''); } function renderProductDetail(products){ const root = document.getElementById('detailRoot'); if(!root) return; const params = new URLSearchParams(location.search); const id = params.get('id'); const p = products.find(x => x.id === id) || products[0]; if(!p){ root.innerHTML = '
No product found.
'; return; } const isOriginal = (p.type || '').toLowerCase() === 'original'; root.innerHTML = `

${p.name}

${p.category}${isOriginal && p.artist ? ' · ' + p.artist : ''}

${moneyZAR(p.price)}

${isOriginal ? `

${p.medium ? 'Medium: ' + p.medium + '
' : ''} ${p.edition ? 'Edition: ' + p.edition + '
' : ''} ${p.signed ? 'Signed: Yes
' : ''}

Enquire / Reserve (Demo) ` : ` `}
`; // cart panel wiring if exists wireCartPanel(products); updateCartCount(); renderCartPanel(products); } function wireCartPanel(products){ const panel = document.getElementById('cartPanel'); const openBtn = document.getElementById('cartButton'); const closeBtn = document.getElementById('closeCart'); if(openBtn && panel) openBtn.onclick = () => panel.classList.add('open'); if(closeBtn && panel) closeBtn.onclick = () => panel.classList.remove('open'); // qty buttons delegated document.addEventListener('click', (e) => { const btn = e.target.closest('[data-qty]'); if(!btn) return; const id = btn.getAttribute('data-id'); const delta = Number(btn.getAttribute('data-qty')); updateQty(id, delta); }); } function renderCartPanel(products){ const itemsEl = document.getElementById('cartItems'); const totalEl = document.getElementById('cartTotal'); if(!itemsEl || !totalEl) return; const cart = readCart(); if(!cart.length){ itemsEl.innerHTML = '
Your cart is empty.
'; totalEl.textContent = moneyZAR(0); return; } let total = 0; itemsEl.innerHTML = cart.map(ci => { const p = products.find(x => x.id === ci.id); if(!p) return ''; const line = (Number(p.price)||0) * ci.qty; total += line; return `
${p.name}
${moneyZAR(p.price)}
Qty ${ci.qty}
${moneyZAR(line)}
`; }).join(''); totalEl.textContent = moneyZAR(total); } function renderCheckout(products){ const cartEl = document.getElementById('cart'); const totalEl = document.getElementById('checkoutTotal'); if(!cartEl || !totalEl) return; const cart = readCart(); if(!cart.length){ cartEl.innerHTML = '
Your cart is empty. Return to the shop and add items.
'; totalEl.textContent = moneyZAR(0); return; } let total = 0; cartEl.innerHTML = cart.map(ci => { const p = products.find(x => x.id === ci.id); if(!p) return ''; const line = (Number(p.price)||0) * ci.qty; total += line; return `
${p.name}
${p.category} · Qty ${ci.qty}
${moneyZAR(line)}
`; }).join(''); totalEl.textContent = moneyZAR(total); const payBtn = document.getElementById('payNow'); if(payBtn){ payBtn.onclick = () => alert('Demo checkout: ready for PayFast / Peach Payments integration.'); } } async function init(){ updateCartCount(); const products = await loadProducts(); // Homepage if(document.getElementById('products')){ wireNavFilters(products); renderHome(products, 'All'); } // Product detail if(document.getElementById('detailRoot')){ renderProductDetail(products); } // Checkout if(document.getElementById('checkoutTotal')){ renderCheckout(products); } } document.addEventListener('DOMContentLoaded', () => { init().catch(err => { console.error(err); const grid = document.getElementById('products'); if(grid) grid.innerHTML = '
Could not load products. Ensure products.json is in the repo root.
'; }); });
